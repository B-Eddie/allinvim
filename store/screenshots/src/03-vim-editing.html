// Esc leaves insert — now every key is a Vim command.
const renameSymbol = () => {
  const word = readWordUnderCursor();
  // ciw — change inner word, stay in flow.
  applyEdit(word, 'newName');
  return true;
};

// Motions: w b e 0 ^ $ gg G f F t T — everywhere you type:
// textareas, inputs, Google Docs, code editors.
function repeatLastChange() {
  dotRepeat(); // the . command, on the web
}

// Operators compose with motions: daw — delete a word,
// ci" — change inside quotes, yap — yank a paragraph.
const motions = ['w', 'b', 'e', '0', '^', '$'];
const operators = ['d', 'c', 'y', 'v', 'V'];
applyEdit(motions, operators); // fluent everywhere
